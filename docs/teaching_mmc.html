Outil interactif

Entrez les composantes du tenseur des contraintes (en MPa), en tapant la valeur dans la case ou en déplaçant le curseur.

viewof sxx = Inputs.range([-300, 300], {value: 40, step: 1, label: "σxx (MPa)"})
viewof syy = Inputs.range([-300, 300], {value: -40, step: 1, label: "σyy (MPa)"})
viewof sxy = Inputs.range([-300, 300], {value: 30, step: 1, label: "σxy (MPa)"})
c = (sxx + syy) / 2
R = Math.sqrt(((sxx - syy) / 2) ** 2 + sxy ** 2)
s1 = c + R
s2 = c - R
theta = 0.5 * Math.atan2(2 * sxy, sxx - syy)
thetaDeg = theta * 180 / Math.PI
e1 = [Math.cos(theta), Math.sin(theta)]
e2 = [-Math.sin(theta), Math.cos(theta)]

Cercle de Mohr

{
  const cercle = d3.range(0, 361).map(a => ({
    s: c + R * Math.cos(a * Math.PI / 180),
    t: R * Math.sin(a * Math.PI / 180)
  }));
  const X = {s: sxx, t: -sxy, label: "X (facette x)"};
  const Y = {s: syy, t: sxy, label: "Y (facette y)"};
  const principaux = [
    {s: s1, t: 0, label: "σI"},
    {s: s2, t: 0, label: "σII"}
  ];
  const m = Math.max(R, 1) * 1.3;
  return Plot.plot({
    width: 500, height: 500,
    x: {domain: [c - m, c + m], label: "σn (MPa)"},
    y: {domain: [-m, m], label: "τn (MPa)"},
    grid: true,
    marks: [
      Plot.ruleX([0], {clip: true}),
      Plot.ruleY([0]),
      Plot.line(cercle, {x: "s", y: "t", stroke: "#5c8037", strokeWidth: 3}),
      Plot.line([X, Y], {x: "s", y: "t", stroke: "black"}),
      Plot.dot([X, Y], {x: "s", y: "t", fill: "black", r: 4}),
      Plot.dot(principaux, {x: "s", y: "t", fill: "red", r: 5}),
      Plot.dot([{s: c, t: 0}], {x: "s", y: "t", fill: "white", stroke: "black"}),
      Plot.text([X, Y, ...principaux], {x: "s", y: "t", text: "label",
                 dx: 8, dy: -12, textAnchor: "start", fontSize: 14})
    ]
  });
}

Directions principales dans le plan (x, y)

{
  const fl1 = [{x1: 0, y1: 0, x2: e1[0], y2: e1[1]},
               {x1: 0, y1: 0, x2: -e1[0], y2: -e1[1]}];
  const fl2 = [{x1: 0, y1: 0, x2: e2[0], y2: e2[1]},
               {x1: 0, y1: 0, x2: -e2[0], y2: -e2[1]}];
  const a = 0.4;
  const element = [[-a,-a],[a,-a],[a,a],[-a,a],[-a,-a]].map(([u, v]) => ({
    x: u * Math.cos(theta) - v * Math.sin(theta),
    y: u * Math.sin(theta) + v * Math.cos(theta)
  }));
  const labels = [
    {x: 1.3 * e1[0], y: 1.3 * e1[1], t: `I (σI = ${s1.toFixed(1)})`, col: "red"},
    {x: 1.3 * e2[0], y: 1.3 * e2[1], t: `II (σII = ${s2.toFixed(1)})`, col: "blue"}
  ];
  return Plot.plot({
    width: 500, height: 500,
    x: {domain: [-1.6, 1.6], label: "x"},
    y: {domain: [-1.6, 1.6], label: "y"},
    grid: true,
    marks: [
      Plot.ruleX([0]), Plot.ruleY([0]),
      Plot.line(element, {x: "x", y: "y", stroke: "gray", strokeDasharray: "4,3"}),
      Plot.arrow(fl1, {x1: "x1", y1: "y1", x2: "x2", y2: "y2", stroke: "red", strokeWidth: 2.5}),
      Plot.arrow(fl2, {x1: "x1", y1: "y1", x2: "x2", y2: "y2", stroke: "blue", strokeWidth: 2.5}),
      Plot.text(labels.slice(0, 1), {x: "x", y: "y", text: "t", fill: "red", fontSize: 13}),
      Plot.text(labels.slice(1), {x: "x", y: "y", text: "t", fill: "blue", fontSize: 13})
    ]
  });
}

Le carré en pointillés représente l’élément de matière orienté selon les directions principales : sur ses faces, le cisaillement est nul.
