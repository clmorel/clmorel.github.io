Exercice - cercle de Mohr interactif

Tenseur des contraintes (MPa) : tapez une valeur ou déplacez le curseur.

viewof sxx = Inputs.range([-300, 300], {value: 40, step: 1, label: html`σ<sub>xx</sub>`})
viewof syy = Inputs.range([-300, 300], {value: -40, step: 1, label: html`σ<sub>yy</sub>`})
viewof sxy = Inputs.range([-300, 300], {value: 30, step: 1, label: html`σ<sub>xy</sub>`})
c = (sxx + syy) / 2
R = Math.sqrt(((sxx - syy) / 2) ** 2 + sxy ** 2)
s1 = c + R
s2 = c - R
theta = 0.5 * Math.atan2(2 * sxy, sxx - syy)
thetaDeg = theta * 180 / Math.PI
e1 = [Math.cos(theta), Math.sin(theta)]
e2 = [-Math.sin(theta), Math.cos(theta)]
couleurs = ({bleu: "#3b6ea5", orange: "#d98c2b", violet: "#7b5ea7", gris: "#8a94a0"})

Cercle de Mohr

{
  const deg = Math.PI / 180;
  const cercle = d3.range(0, 361).map(a => ({s: c + R * Math.cos(a * deg), t: R * Math.sin(a * deg)}));
  const angX = Math.atan2(-sxy, sxx - c);
  const arc = d3.range(0, 41).map(i => {
    const a = angX * i / 40;
    return {s: c + 0.3 * R * Math.cos(a), t: 0.3 * R * Math.sin(a)};
  });
  const X = {s: sxx, t: -sxy, label: "X"};
  const Y = {s: syy, t: sxy, label: "Y"};
  const P = [{s: s1, t: 0, label: `σI = ${s1.toFixed(1)}`, dx: 6, anchor: "start"},
             {s: s2, t: 0, label: `σII = ${s2.toFixed(1)}`, dx: -6, anchor: "end"}];
  const T = [{s: c, t: R, label: "τmax"}, {s: c, t: -R, label: ""}];
  const m = Math.max(R, 1) * 1.45;
  return Plot.plot({
    width: 420, height: 420,
    x: {domain: [c - m, c + m], label: "σn (MPa) →"},
    y: {domain: [-m, m], label: "↑ τn (MPa)"},
    grid: true,
    marks: [
      Plot.ruleY([0], {stroke: couleurs.gris}),
      Plot.ruleX([0], {stroke: couleurs.gris, clip: true}),
      Plot.line(cercle, {x: "s", y: "t", stroke: couleurs.bleu, strokeWidth: 3}),
      Plot.line([X, Y], {x: "s", y: "t", stroke: couleurs.gris, strokeDasharray: "5,4"}),
      Plot.line(arc, {x: "s", y: "t", stroke: couleurs.orange, strokeWidth: 2}),
      Plot.text([{s: c + 0.42 * R * Math.cos(angX / 2), t: 0.42 * R * Math.sin(angX / 2)}],
                {x: "s", y: "t", text: () => "2θ", fill: couleurs.orange, fontSize: 13, fontWeight: "bold"}),
      Plot.dot([{s: c, t: 0}], {x: "s", y: "t", r: 4, fill: "white", stroke: "currentColor"}),
      Plot.dot(T, {x: "s", y: "t", r: 4, fill: couleurs.violet}),
      Plot.text(T, {x: "s", y: "t", text: "label", dy: -12, fill: couleurs.violet, fontSize: 12}),
      Plot.dot([X, Y], {x: "s", y: "t", r: 5, fill: "currentColor"}),
      Plot.text([X, Y], {x: "s", y: "t", text: "label", dx: 9, dy: -9, fontSize: 14, fontWeight: "bold"}),
      Plot.dot(P, {x: "s", y: "t", r: 6, fill: couleurs.orange}),
      Plot.text(P, {x: "s", y: "t", text: "label", dx: "dx", dy: 14, textAnchor: "anchor",
                    fill: couleurs.orange, fontSize: 12, fontWeight: "bold"})
    ]
  });
}

Directions principales

{
  const r2 = Math.SQRT1_2;
  const t1 = [r2 * (e1[0] + e2[0]), r2 * (e1[1] + e2[1])];   // cisaillement max (+45°)
  const t2 = [r2 * (e1[0] - e2[0]), r2 * (e1[1] - e2[1])];   // cisaillement max (−45°)
  const smax = Math.max(Math.abs(s1), Math.abs(s2), 1e-9);

  const axe = (n) => [{x: -1.7 * n[0], y: -1.7 * n[1]}, {x: 1.7 * n[0], y: 1.7 * n[1]}];
  // flèches depuis l'origine : sortantes en traction, rentrantes en compression
  const fleches = (n, s) => [1, -1].map(k => {
    const L = 0.35 + 0.75 * Math.abs(s) / smax;
    const p = [k * 0.1 * n[0], k * 0.1 * n[1]];
    const q = [k * L * n[0], k * L * n[1]];
    return s >= 0 ? {x1: p[0], y1: p[1], x2: q[0], y2: q[1]}
                  : {x1: q[0], y1: q[1], x2: p[0], y2: p[1]};
  });
  const arcT = d3.range(0, 31).map(i => ({x: 0.3 * Math.cos(theta * i / 30),
                                          y: 0.3 * Math.sin(theta * i / 30)}));
  const lab = (n, txt, col) => ({x: 1.5 * n[0], y: 1.5 * n[1], t: txt, col});

  return Plot.plot({
    width: 420, height: 420,
    x: {domain: [-1.7, 1.7], label: "x →"},
    y: {domain: [-1.7, 1.7], label: "↑ y"},
    grid: true,
    marks: [
      Plot.ruleX([0], {stroke: couleurs.gris}), Plot.ruleY([0], {stroke: couleurs.gris}),
      Plot.line(axe(e1), {x: "x", y: "y", stroke: couleurs.orange, strokeWidth: 1.5}),
      Plot.line(axe(e2), {x: "x", y: "y", stroke: couleurs.bleu, strokeWidth: 1.5}),
      Plot.line(axe(t1), {x: "x", y: "y", stroke: couleurs.violet, strokeWidth: 1.5, strokeDasharray: "6,4"}),
      Plot.line(axe(t2), {x: "x", y: "y", stroke: couleurs.violet, strokeWidth: 1.5, strokeDasharray: "6,4"}),
      Plot.arrow(fleches(e1, s1), {x1: "x1", y1: "y1", x2: "x2", y2: "y2",
                 stroke: couleurs.orange, strokeWidth: 3, headLength: 10}),
      Plot.arrow(fleches(e2, s2), {x1: "x1", y1: "y1", x2: "x2", y2: "y2",
                 stroke: couleurs.bleu, strokeWidth: 3, headLength: 10}),
      Plot.line(arcT, {x: "x", y: "y", stroke: "currentColor", strokeWidth: 1.5}),
      Plot.text([{x: 0.42 * Math.cos(theta / 2), y: 0.42 * Math.sin(theta / 2)}],
                {x: "x", y: "y", text: () => "θp", fontSize: 12, fontWeight: "bold"}),
      Plot.text([lab(e1, `I · σI = ${s1.toFixed(1)}`, couleurs.orange),
                 lab(e2, `II · σII = ${s2.toFixed(1)}`, couleurs.bleu),
                 lab(t1, "τmax", couleurs.violet),
                 lab(t2, "τmax", couleurs.violet)],
                {x: "x", y: "y", text: "t", fill: "col", fontSize: 12, fontWeight: "bold"})
    ]
  });
}

Les droites pleines sont les directions principales (flèches sortantes : traction, rentrantes : compression). Les droites violettes en pointillés sont les directions de cisaillement maximal, à 45° des directions principales.
